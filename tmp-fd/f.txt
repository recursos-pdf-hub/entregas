/* ---- Countdown to local midnight ---- */
(function(){
  var inline=document.querySelector('[data-timer-inline]');
  var h=document.querySelector('[data-t="h"]'),m=document.querySelector('[data-t="m"]'),s=document.querySelector('[data-t="s"]');
  function pad(n){return String(n).padStart(2,'0')}
  function tick(){
    var now=new Date(),end=new Date(now);end.setHours(23,59,59,999);
    var d=Math.max(0,Math.floor((end-now)/1000));
    var hh=pad(Math.floor(d/3600)),mm=pad(Math.floor(d%3600/60)),ss=pad(d%60);
    if(inline)inline.textContent=hh+':'+mm+':'+ss;
    if(h){h.textContent=hh;m.textContent=mm;s.textContent=ss;}
  }
  tick();setInterval(tick,1000);
})();
