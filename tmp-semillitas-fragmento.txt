<script>
  // ── STORY CAROUSEL ──
  const storyState = {};   // { carouselId: { index, timer } }
  const STORY_DURATION = 4000; // ms por slide no avanço automático

  function initStoryCarousel(carousel) {
    const id = carousel.id;
    storyState[id] = { index: 0, timer: null };
    updateStoryCarousel(id);
    startStoryTimer(id);

    // pausa o avanço automático enquanto o usuário interage
    carousel.addEventListener('mouseenter', () => stopStoryTimer(id));
    carousel.addEventListener('mouseleave', () => startStoryTimer(id));
  }

  function updateStoryCarousel(id) {
    const carousel = document.getElementById(id);
    const state = storyState[id];
    const track = carousel.querySelector('.sf-story-track');
    const dots = carousel.parentElement.querySelectorAll('.sf-story-dot');

    track.style.transform = `translateX(-${state.index * 100}%)`;
    dots.forEach((d, i) => d.classList.toggle('sf-active', i === state.index));
  }

  function storyNav(id, direction) {
    const carousel = document.getElementById(id);
    const total = carousel.querySelectorAll('.sf-story-slide').length;
    const state = storyState[id];
    state.index = (state.index + direction + total) % total;
    updateStoryCarousel(id);
    startStoryTimer(id);
  }

  function storyGo(id, index) {
    storyState[id].index = index;
    updateStoryCarousel(id);
    startStoryTimer(id);
  }

  function startStoryTimer(id) {
    stopStoryTimer(id);
    storyState[id].timer = setTimeout(() => storyNav(id, 1), STORY_DURATION);
  }

  function stopStoryTimer(id) {
    if (storyState[id] && storyState[id].timer) {
      clearTimeout(storyState[id].timer);
    }
  }

  document.querySelectorAll('.sf-story-carousel').forEach(initStoryCarousel);

  // ── FAQ TOGGLE ──
  function toggleFaq(el) {
    el.classList.toggle('sf-open');
    const ans = el.nextElementSibling;
    ans.classList.toggle('sf-open');
  }

  // ── DATA DE HOJE NA OFERTA ──
  (function() {
    const el = document.getElementById('sf-data-hoje');
    if (!el) return;
    el.textContent = new Date().toLocaleDateString('es-AR', {
      day: 'numeric', month: 'long', year: 'numeric'
    });
  })();



  // ── SMOOTH SCROLL ──
  document.querySelectorAll('a[href^="#"]').forEach(link => {
    link.addEventListener('click', function(e) {
      const target = document.querySelector(this.getAttribute('href'));
      if (target) {
        e.preventDefault();
        var rm = false; try { rm = window.matchMedia('(prefers-reduced-motion: reduce)').matches; } catch (err) {}
        target.scrollIntoView({ behavior: rm ? 'auto' : 'smooth', block: 'start' });
      }
    });
  });
</script>
